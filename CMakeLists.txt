cmake_minimum_required(VERSION 3.16)
project(hdd-noise VERSION 0.1.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
if(NOT WIN32)
    message(FATAL_ERROR "The application uses Windows audio and disk counters. Synth.cpp is portable.")
endif()
option(HDD_AUDIO_INTEGRATION_TESTS "Test the real default audio device (muted)" OFF)
if(BUILD_TESTING AND HDD_AUDIO_INTEGRATION_TESTS)
    add_executable(AudioEngineTests tests/AudioEngineTests.cpp AudioEngine.cpp Synth.cpp)
    target_include_directories(AudioEngineTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(AudioEngineTests PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(AudioEngineTests PRIVATE winmm)
    if(MSVC)
        target_compile_options(AudioEngineTests PRIVATE /utf-8 /W4)
    endif()
    add_test(NAME audio_integration COMMAND AudioEngineTests)
    set_tests_properties(audio_integration PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 20)
endif()
option(HDD_DESKTOP_INTEGRATION_TESTS "Test embedded icons and actual Explorer tray (no sound)" OFF)
if(BUILD_TESTING AND HDD_DESKTOP_INTEGRATION_TESTS)
    add_executable(TrayTests tests/TrayTests.cpp Synth.cpp AudioEngine.cpp DiskMonitor.cpp App.rc)
    target_include_directories(TrayTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(TrayTests PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(TrayTests PRIVATE user32 gdi32 shell32 comctl32 winmm pdh advapi32)
    if(MSVC)
        target_compile_options(TrayTests PRIVATE /utf-8 /W4)
        target_link_options(TrayTests PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
    endif()
    add_test(NAME desktop_integration COMMAND TrayTests)
    set_tests_properties(desktop_integration PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 20)
endif()
add_executable(hdd-noise WIN32 App.cpp Synth.cpp AudioEngine.cpp DiskMonitor.cpp App.rc)
target_compile_definitions(hdd-noise PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(hdd-noise PRIVATE user32 gdi32 shell32 comctl32 winmm pdh)
if(MSVC)
    target_compile_options(hdd-noise PRIVATE /utf-8 /W4)
    target_link_options(hdd-noise PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest")
    set_property(TARGET hdd-noise PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(MINGW)
    target_sources(hdd-noise PRIVATE MinGWManifest.rc)
    target_link_options(hdd-noise PRIVATE -municode -static)
endif()
if(BUILD_TESTING)
    add_executable(SynthTests tests/SynthTests.cpp Synth.cpp)
    target_include_directories(SynthTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    if(MSVC)
        target_compile_options(SynthTests PRIVATE /utf-8 /W4)
    endif()
    add_test(NAME synthesis COMMAND SynthTests)
endif()
